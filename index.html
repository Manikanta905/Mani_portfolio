// LI-FI TRANSMITTER — Arduino UNO

#define LED_PIN 9

void setup() {
  pinMode(LED_PIN, OUTPUT);
  Serial.begin(9600);
}

void sendBit(int bitValue) {
  if (bitValue == 1) {
    digitalWrite(LED_PIN, HIGH);
  } else {
    digitalWrite(LED_PIN, LOW);
  }
  delay(10);
}

void sendCharacter(char data) {
  // Start bit
  sendBit(0);
  // Send 8 data bits (LSB first)
  for (int i = 0; i < 8; i++) {
    sendBit((data >> i) & 1);
  }
  // Stop bit
  sendBit(1);
}

void loop() {
  String message = "HELLO LIFI";
  Serial.println("Transmitting: " + message);
  for (int i = 0; i < message.length(); i++) {
    sendCharacter(message[i]);
  }
  delay(1000);
}
LI-FI RECEIVER — Arduino UNO 
// LI-FI RECEIVER — Arduino UNO

#define LDR_PIN   A0
#define THRESHOLD 500

const int BIT_DELAY = 10;

void setup() {
  Serial.begin(9600);
}

void loop() {
  int sensorValue = analogRead(LDR_PIN);

  // Detect start bit (light OFF)
  if (sensorValue < THRESHOLD) {
    delay(BIT_DELAY + BIT_DELAY / 2);

    byte receivedData = 0;

    // Receive 8 bits
    for (int i = 0; i < 8; i++) {
      int value = analogRead(LDR_PIN);
      if (value > THRESHOLD) {
        receivedData |= (1 << i);
      }
      delay(BIT_DELAY);
    }

    // Stop bit
    delay(BIT_DELAY);
    Serial.print((char)receivedData);
  }
}
